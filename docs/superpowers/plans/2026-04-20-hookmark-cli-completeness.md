export interface Positional {
  name: string;
  optional: boolean;
  variadic: boolean;
}

// Parse positional tokens from a usage line. Bracketed tokens ([FILE]) are
// optional; bare uppercase tokens are required. A trailing `...` marks the
// token as variadic (one or more). Order in the usage string is preserved.
export function extractPositionals(usage: string): Positional[] {
  const matches = usage.match(/\[[A-Z_]{2,}\](?:\.\.\.)?|\b[A-Z_]{2,}(?:\.\.\.)?/g) ?? [];
  return matches.map((tok) => {
    const variadic = tok.endsWith("...");
    const stripped = variadic ? tok.slice(0, -3) : tok;
    return {
      name: stripped.replace(/^\[|\]$/g, ""),
      optional: stripped.startsWith("["),
      variadic,
    };
  });
}
```

export function regenHookSpec(opts: {
  hookCli: string;
  specPath: string;
  previousPath: string;
}): RegenResult {
  const subcommands: HookSubcommand[] = [];

  for (const sub of SUBCOMMANDS) {
    const help = runHookHelp(opts.hookCli, sub);
    subcommands.push(parseHookHelpOutput(sub, help));
  }

  for (const ss of SUB_SUBCOMMANDS) {
    const help = runHookHelp(opts.hookCli, ss.cliArgs.join(" "));
    const parsed = parseHookHelpOutput(ss.name, help);
    subcommands.push({ ...parsed, cliArgs: [...ss.cliArgs] });
  }

  const spec: HookCliSpec = {
    generatedAt: new Date().toISOString(),
    hookVersion: runHookVersion(opts.hookCli),
    subcommands,
  };
  const newRaw = JSON.stringify(spec, null, 2) + "\n";

  const previousRaw = existsSync(opts.specPath) ? readFileSync(opts.specPath, "utf8") : null;
  const changed = previousRaw !== null && normalizeForDiff(previousRaw) !== normalizeForDiff(newRaw);

  mkdirSync(path.dirname(opts.specPath), { recursive: true });
  if (changed && previousRaw) {
    writeFileSync(opts.previousPath, previousRaw);
  }
  writeFileSync(opts.specPath, newRaw);

  return { spec, changed, previousRaw, newRaw };
}
```

const META_SUBCOMMANDS = new Set(["help", "scripts", "percent"]);

export function discoverCanonicalSubcommands(topLevelHelp: string): string[] {
  const sections = (() => {
    const out: Record<string, string[]> = {};
    let current: string | null = null;
    for (const line of topLevelHelp.split("\n")) {
      const header = line.match(/^([A-Z][A-Z ]{1,30}[A-Z])\s*$/);
      if (header) {
        current = header[1].trim();
        out[current] = [];
      } else if (current) {
        out[current].push(line);
      }
    }
    return out;
  })();

  const lines = sections.COMMANDS ?? [];
  const result: string[] = [];
  for (const raw of lines) {
    const m = raw.trim().match(/^([a-z_][a-z_0-9]*)\s*(?:,\s*[a-z_][a-z_0-9]*)?\s*-/);
    if (!m) continue;
    const canonical = m[1];
    if (META_SUBCOMMANDS.has(canonical)) continue;
    if (canonical.startsWith("_")) continue;
    result.push(canonical);
  }
  return result;
}
```

export function extractAliases(topLevelHelp: string): Record<string, string[]> {
  const result: Record<string, string[]> = {};
  const match = topLevelHelp.match(/\nCOMMANDS\s*\n([\s\S]*?)(?=\n[A-Z][A-Z ]{1,30}[A-Z]\s*\n|\s*$)/);
  const body = match ? match[1] : "";
  for (const raw of body.split("\n")) {
    const m = raw.trim().match(/^([a-z_][a-z_0-9]*)(?:,\s*([a-z_][a-z_0-9, ]*))?\s+-/);
    if (!m) continue;
    const canonical = m[1];
    const rest = m[2] ? m[2].split(",").map((s) => s.trim()).filter(Boolean) : [];
    result[canonical] = rest;
  }
  return result;
}
```

Every data-touching `hook` subcommand (`clip`, `link`, `list`, `find`, `remove`,

## Out of scope (deliberately deferred)

- **URL sanitization before forwarding to hook** — the hookapp gem does unescaped AppleScript interpolation (finding #7). We should reject clearly malformed URLs, but settling on the right policy (allowlist vs. denylist, escaping rules) deserves its own spec.
- **stdin pass-through** for `percent encode -` style invocations (finding #8). Our `runHook` wrapper currently uses `execSync`; streaming stdin through HTTP is a larger change.
- **`scripts <SHELL>` subcommand** — purely emits shell completion examples; no obvious agent use case.
- **`_doc` subcommand** — GLI built-in, returns empty output for the only supported format.
